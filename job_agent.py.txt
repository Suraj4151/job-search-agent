# job_agent.py
import os
import io
import time
import json
import smtplib
import requests
import pandas as pd
from bs4 import BeautifulSoup
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
import gspread
from oauth2client.service_account import ServiceAccountCredentials
from datetime import datetime

# Config
EXCEL_PATH = os.getenv("EXCEL_PATH", "Consolidated_Corporate_Directory_v2.xlsx")
TARGET_TITLES = [
    "project manager", "program manager", "business analyst",
    "product manager", "systems analyst", "product owner",
    "pm", "pmo", "implementation manager", "technical business analyst"
]
KEYWORDS = [
    "medicare", "medicaid", "dsnp", "pmo", "raid", "uat", "sdlc",
    "agile", "scrum", "requirements", "transformation", "modernization",
    "excel", "sql", "power bi", "powerbi"
]

# Email config from env
EMAIL_USER = os.getenv("EMAIL_USER")
EMAIL_PASSWORD = os.getenv("EMAIL_PASSWORD")
EMAIL_TO = os.getenv("EMAIL_TO", EMAIL_USER)

# Google Sheets setup
GCP_JSON = os.getenv("GCP_SERVICE_ACCOUNT_JSON")
SHEET_ID = os.getenv("SHEET_ID")

def load_companies(excel_path):
    xls = pd.ExcelFile(excel_path)
    companies = []
    for sheet in xls.sheet_names:
        df = xls.parse(sheet)
        # Try to find columns that look like company and careers url
        cols = [c.lower() for c in df.columns]
        # heuristics
        if "company" in cols:
            company_col = df.columns[cols.index("company")]
        elif "company name" in cols:
            company_col = df.columns[cols.index("company name")]
        else:
            company_col = df.columns[0]
        url_col = None
        for candidate in ["careers url", "careers", "careers url "]:
            if candidate in cols:
                url_col = df.columns[cols.index(candidate)]
                break
        # fallback: try to find any column with 'http' in values
        if url_col is None:
            for c in df.columns:
                if df[c].astype(str).str.contains("http").any():
                    url_col = c
                    break
        for _, row in df.iterrows():
            name = str(row.get(company_col, "")).strip()
            url = str(row.get(url_col, "")).strip() if url_col is not None else ""
            if name:
                companies.append({"company": name, "url": url, "domain": sheet})
    return companies

def fetch_job_links_from_page(url):
    try:
        r = requests.get(url, timeout=15)
        r.raise_for_status()
    except Exception:
        return []
    soup = BeautifulSoup(r.text, "html.parser")
    links = []
    # common patterns: job links often contain '/jobs', '/careers', '/job/'
    for a in soup.find_all("a", href=True):
        href = a["href"]
        text = a.get_text(separator=" ", strip=True)
        if any(k in href.lower() for k in ["/job", "/jobs", "/careers", "/careers/"] ) or any(t in text.lower() for t in TARGET_TITLES):
            full = href if href.startswith("http") else requests.compat.urljoin(url, href)
            links.append({"title": text, "url": full})
    # dedupe
    seen = set()
    out = []
    for l in links:
        if l["url"] not in seen:
            seen.add(l["url"])
            out.append(l)
    return out

def fetch_job_description(url):
    try:
        r = requests.get(url, timeout=15)
        r.raise_for_status()
    except Exception:
        return ""
    soup = BeautifulSoup(r.text, "html.parser")
    # heuristics: job descriptions often in <div> with class containing 'description' or <section>
    text = ""
    for selector in ["div[class*='description']", "div[class*='job']", "section", "article", "div"]:
        el = soup.select_one(selector)
        if el:
            text = el.get_text(separator=" ", strip=True)
            if len(text) > 200:
                break
    if not text:
        text = soup.get_text(separator=" ", strip=True)
    return text

def score_match(title, description):
    title_lower = title.lower()
    desc_lower = description.lower()
    title_score = sum(1 for t in TARGET_TITLES if t in title_lower)
    keyword_hits = sum(1 for k in KEYWORDS if k in desc_lower)
    score = title_score * 2 + keyword_hits
    return min(100, score * 10)  # scale to 0-100

def process_companies(companies, max_per_company=10):
    rows = []
    for c in companies:
        url = c.get("url") or ""
        if not url:
            continue
        links = fetch_job_links_from_page(url)
        # limit to first N links to avoid heavy scraping
        for link in links[:max_per_company]:
            title = link.get("title") or ""
            job_url = link.get("url")
            desc = fetch_job_description(job_url) if job_url else ""
            score = score_match(title, desc)
            # filter: only keep if score >= threshold or title contains target
            if score >= 30 or any(t in title.lower() for t in TARGET_TITLES):
                rows.append({
                    "timestamp": datetime.utcnow().isoformat(),
                    "company": c.get("company"),
                    "domain": c.get("domain"),
                    "title": title,
                    "location": "", 
                    "score": score,
                    "url": job_url,
                    "snippet": (desc[:400] + "...") if desc else ""
                })
        time.sleep(1)  # polite pause
    return rows

def append_to_google_sheet(rows):
    if not GCP_JSON or not SHEET_ID:
        print("Google Sheets not configured.")
        return
    creds_dict = json.loads(GCP_JSON)
    scope = ["https://spreadsheets.google.com/feeds","https://www.googleapis.com/auth/drive"]
    creds = ServiceAccountCredentials.from_json_keyfile_dict(creds_dict, scope)
    client = gspread.authorize(creds)
    sheet = client.open_by_key(SHEET_ID).sheet1
    header = ["timestamp","company","domain","title","location","score","url","snippet"]
    existing = sheet.row_values(1)
    if not existing:
        sheet.append_row(header)
    for r in rows:
        row = [r.get(h,"") for h in header]
        sheet.append_row(row)

def send_email(rows):
    if not EMAIL_USER or not EMAIL_PASSWORD:
        print("Email not configured.")
        return
    # build HTML table for top 20 rows
    top = sorted(rows, key=lambda x: x["score"], reverse=True)[:20]
    html = "<h3>Daily Job Matches</h3><table border='1' cellpadding='4' cellspacing='0'><tr><th>Company</th><th>Title</th><th>Domain</th><th>Score</th><th>Link</th></tr>"
    for r in top:
        html += f"<tr><td>{r['company']}</td><td>{r['title']}</td><td>{r['domain']}</td><td>{r['score']}</td><td><a href='{r['url']}'>link</a></td></tr>"
    html += "</table>"
    msg = MIMEMultipart("alternative")
    msg["Subject"] = "Daily Job Matches — Suraj"
    msg["From"] = EMAIL_USER
    msg["To"] = EMAIL_TO
    part = MIMEText(html, "html")
    msg.attach(part)
    # send via SMTP
    server = smtplib.SMTP_SSL("smtp.gmail.com", 465)
    server.login(EMAIL_USER, EMAIL_PASSWORD)
    server.sendmail(EMAIL_USER, [EMAIL_TO], msg.as_string())
    server.quit()

def main():
    companies = load_companies(EXCEL_PATH)
    print(f"Loaded {len(companies)} companies.")
    rows = process_companies(companies)
    if rows:
        append_to_google_sheet(rows)
        send_email(rows)
    else:
        print("No matches found today.")

if __name__ == "__main__":
    main()
